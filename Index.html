// Pastikan chat-display kosong dulu const chatDisplay = document.getElementById("chat-display"); function sendMessage() { let input = document.getElementById("user-input").value; // Reset tampilan sebelum menerima respon baru chatDisplay.innerHTML = '|'; fetch('/eve', { method: 'POST', body: JSON.stringify({ message: input }), headers: { 'Content-Type': 'application/json' } }) .then(response => response.json()) .then(data => { // Panggil fungsi typeWriter dengan teks dari server typeWriter(data.reply, 0); }); } function typeWriter(text, i) { if (i < text.length) { // Menggunakan substring untuk menambah karakter satu-satu // Kita pakai text.charAt(i) biar lebih aman untuk karakter khusus seperti \n chatDisplay.innerHTML = text.substring(0, i + 1) + '|'; // Kecepatan diatur lebih cepat (10ms) agar kotak-kotak tidak nunggu lama setTimeout(function() { typeWriter(text, i + 1); }, 10); } else { // Hapus kursor setelah selesai mengetik chatDisplay.innerHTML = text; } }
